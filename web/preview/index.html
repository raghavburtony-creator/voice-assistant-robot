Use it

<script type="module" src="/otto-mascot.js"></script>
<otto-mascot theme="midnight" hotkey="O"></otto-mascot>

<script type="module">
  const otto = document.querySelector('otto-mascot')
  otto.addEventListener('otto:ready', () => otto.playGesture('welcomeWave'))
  otto.addEventListener('otto:activate', () => openMyChat())
  // Say what happened; Otto decides how to react.
  otto.notify({ type: 'success', where: '#save', topic: 'upload' })
  const job = otto.task('render', { topic: 'image generation', where: '#preview' })
  job.succeed('Your image is ready')
</script>
// Without the element: any container.
import { createOtto } from 'otto-mascot'
const otto = createOtto(document.body, { config: { theme: 'cyber', sound: { soundVolume: 0.3 } } })
await otto.moveToElement('#cart'); otto.playGesture('celebrate')
otto.connectAudioNode(myTtsSourceNode)   // lip-sync to your own voice
otto.dispose()                           // on route change
// React, Vue or Svelte: it is just an element.
import 'otto-mascot'
<otto-mascot theme="minimal" ref={el} onotto:reaction=... />     // React 19 passes custom events through
<otto-mascot :theme="theme" @otto:gestureend="onEnd" />          // Vue
<otto-mascot theme={theme} on:otto:activate={open} />            // Svelte
